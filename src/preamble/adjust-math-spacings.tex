    % \setlength{\abovedisplayshortskip}{0.1\abovedisplayshortskip}
    \setlength{\abovedisplayshortskip}{0pt} % seemingly no effect
}

    \spaceskip=0.66ex plus 0.1ex minus 0.05ex\relax % words in 'array' env
}
