% Load xcolor with dvipsnames for 68 specific shades
\usepackage[dvipsnames]{xcolor}  % xcolor might get loaded w/ more colors

\newcommand{\linkColorBlue}{MidnightBlue}  % Cerulean, RoyalBue, MidnightBlue
\newcommand{\citeColorRed}{RubineRed}
\newcommand{\boxColorPink}{BurntOrange} % CarnationPink

% \newcommand{\headingFont}{\bfseries\sffamily} % references will also be in this font _shape_
\newcommand{\headingFont}{} % references will also be in this font _shape_
\newcommand{\urlFont}{\sffamily}      % url & href: sans
\newcommand{\urlAltFont}{\ttfamily}   % URL: tt typewriter (monospaced)


% Modify the Section headings (font family and/or color)
\usepackage{sectsty}
\allsectionsfont{\headingFont}  % use \color{\linkColorBlue} to colorize

%%%%% have "Abstract" match the section headings (or not)
\usepackage{abstract}
% \renewcommand{\abstractnamefont}{\headingFont}  %%%% match headingFont ?? %%%%
\renewcommand{\abstractnamefont}{\bf}   %%%% or just match \normalfont (+bold)

\usepackage{hyperref}
\hypersetup{
    colorlinks=false, 
    citebordercolor=\boxColorPink,
    linkbordercolor=\linkColorBlue,
    urlbordercolor=\linkColorBlue,
    pdfborderstyle={/S/S/W 1.2}  % /S: style, /S: solid, /W: width
}

\usepackage[capitalize, noabbrev, nameinlink]{cleveref}

% \cref: "Section 11"
\crefformat{section}{#2{\headingFont\textcolor{\linkColorBlue}{Section~#1}}#3}

% \ref is hard to overwrite, so we use this (labelcref) and overwrite \ref after
\labelcrefformat{section}{#2{\headingFont\textcolor{\linkColorBlue}{#1}}#3}

\AtBeginDocument{ \renewcommand{\ref}{\labelcref} }  % overwrite

% url and link customizations:
\urlstyle{same}

% \url: make 
\let\oldurl\url
\renewcommand{\url}[1]{\mbox{\urlFont\textcolor{\linkColorBlue}{\oldurl{#1}}}}

% \URL: a second URL type in the "typewriter" (i.e. mono) font family
\newcommand{\URL}[1]{%
   \mbox{\href{#1}{{\urlAltFont\textcolor{\linkColorBlue}{\nolinkurl{#1}}}}}%
}

% \href: for LINKS -- where displayed text is not necessarily the url
\let\oldhref\href
\renewcommand{\href}[2]{%
   \oldhref{#1}{\mbox{\urlFont\textcolor{\linkColorBlue}{#2}}}%
}

% \cite: only change the Num inside the brackets at the citing location
\AddToHook{package/biblatex/after}{
   \AtEveryCite{\DeclareFieldFormat{labelnumber}{\textcolor{\citeColorRed}{#1}}}}
