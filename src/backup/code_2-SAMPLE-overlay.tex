\begin{tikzpicture}[>=Triangle]
    \tikzset{
        box/.style={
            draw,
            minimum size=2em,
            text width=3em,
            text centered,
            outer sep=0pt
        },
        container/.style={
            draw,
            inner xsep=1.5em,
            inner ysep=0.75em
        }
    }

% --- STEP 1: BLOCKS AND CONTAINERS (UNTOUCHED) ---
    \foreach[evaluate={\prev=int(\i - 1)}] \i in {1, 2, 3} {
        
        % Internal Content Nodes
        \node (P\i-PubKey) at (\i * 15em, 0) [box, text width=6em] {Owner \i's \\ Public Key};
        \node (P\i-Hash) [below=3em of P\i-PubKey][box] {Hash}; 
        \node (P\i-Sig) [below=2em of P\i-Hash][box, text width=5em] {Owner \prev's \\ Signature};

        % Centered vertical coordinate for horizontal arrows
        \coordinate (P\i-MidY) at ($(P\i-Hash.north)!0.5!(P\i-PubKey.south)$);

        % 1em top padding
        \coordinate (P\i-TopPad) at ($(P\i-PubKey.north) + (0, 1.33em)$);

        % Container (Relative spacing)
        \node (P\i-Tx) [fit=(P\i-TopPad)(P\i-Sig)(P\i-PubKey)][container] {}; 

        % Manual Label 
        \node[anchor=north west, xshift=0.33em, yshift=-0.33em, font=\bfseries] 
            at (P\i-Tx.north west) {Transaction};

        % Internal block arrows
        \draw [->] (P\i-Hash) -- (P\i-Sig);
        \draw [->] (P\i-Hash.45 |- P\i-PubKey.south) -- (P\i-Hash.45);

        % Private Key
        \node (P\i-PrivKey) [below=2.5em of P\i-Tx] [box, text width=6em] {Owner \i's \\ Private Key};
    }

    % --- STEP 2: DYNAMIC BRIDGE LENGTH (UNTOUCHED) ---
    \path let \p1 = (P1-Tx.east), \p2 = (P2-Hash.135 |- P2-Hash.north) in 
      \pgfextra{
        \pgfmathsetmacro{\bridgelen}{\x2-\x1} 
        \global\let\bridgelen\bridgelen
      };

    % Draw incoming arrow
    \draw [->] ([xshift=-\bridgelen pt]P1-Hash.135 |- P1-MidY) -| (P1-Hash.135 |- P1-Hash.north);

    % --- STEP 3: EXTERNAL CONNECTIONS (UNTOUCHED) ---
    \foreach[evaluate={\next=int(\i + 1)}] \i in {1, 2} {
        
        % Verify/Sign dashed lines
        \draw [->, dashed] (P\i-PubKey.330) -- (P\i-PubKey.330 |- P\i-Hash.south east) 
            -- node[sloped, above]{Verify} (P\next-Sig.165);
            
        \draw [->, dashed] (P\i-PrivKey.15) -- node[sloped, above]{Sign} (P\next-Sig.195);
        
        % Block-to-block connection
        \draw [->] (P\i-Tx.east |- P\i-MidY) -| (P\next-Hash.135 |- P\next-Hash.north);
    }

    % --- STEP 4: PHANTOM PADDING FOR CENTERING ---
    % 1. Calculate how far the left tail protrudes past the container wall
    \path let \p1 = (P1-Tx.west), \p2 = ([xshift=-\bridgelen pt]P1-Hash.135) in
        \pgfextra{
            \pgfmathsetmacro{\protrusion}{\x1-\x2}
            \global\let\protrusion\protrusion
        };

    % 2. Reset the Bounding Box to include equal "phantom" width on the right
    \pgfresetboundingbox
    \path[use as bounding box] 
        ([xshift=-\protrusion pt]P1-Tx.west |- P1-Tx.north) 
        rectangle 
        ([xshift=\protrusion pt]P3-Tx.east |- P1-PrivKey.south);
    \node[rotate=23,scale=9.5,opacity=0.25] at (current bounding box.center) {SAMPLE};
    
\end{tikzpicture}
